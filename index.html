// usage

import createFocusTrap from 'focus-trap-tiny';

const modal = document.getElementById('my-modal');
const trap = createFocusTrap(modal, {
  onEscape: () => closeModal(),
});

// When modal opens
trap.activate();

// When modal closes
trap.deactivate();

// If DOM changes dynamically
trap.update();

// features

⌨️

Tab & Shift+Tab

Full bidirectional keyboard cycling through all focusable elements.

🔙

Return Focus

Automatically restores focus to the trigger element when deactivated.

🙈

Inert-aware

Skips elements inside [inert] containers and hidden elements.

Dynamic content
